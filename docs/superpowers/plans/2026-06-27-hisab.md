# Hisab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a full-stack web app for tracking charitable donations in and distributions out, with role-based access control, scoped visibility, and PDF/Excel reporting.

**Architecture:** Next.js 15 App Router (TypeScript) with PostgreSQL + Prisma. All data mutations use Server Actions. Authentication via NextAuth.js v5 credentials provider. Scoping enforced server-side on every Prisma query via a central `lib/permissions.ts` helper.

**Tech Stack:** Next.js 15, PostgreSQL, Prisma 5, NextAuth.js v5, Shadcn/ui, Tailwind CSS, Zod, React Hook Form, @react-pdf/renderer, ExcelJS, Vitest.

**Spec:** `docs/superpowers/specs/2026-06-27-hisab-design.md`

- [ ] **Step 1: Write lib/auth.ts**
```ts
import NextAuth from "next-auth"
import Credentials from "next-auth/providers/credentials"
import { compare } from "bcryptjs"
import { db } from "@/lib/db"
import type { Role, ScopeType } from "@prisma/client"

declare module "next-auth" {
  interface Session {
    user: { id: string; name: string; email: string; role: Role; scopeType: ScopeType }
  }
  interface User { role: Role; scopeType: ScopeType }
}

export const { handlers, auth, signIn, signOut } = NextAuth({
  session: { strategy: "jwt", maxAge: 7 * 24 * 60 * 60 },
  pages: { signIn: "/login" },
  providers: [
    Credentials({
      credentials: { email: {}, password: {} },
      async authorize(credentials) {
        if (!credentials?.email || !credentials?.password) return null
        const user = await db.user.findUnique({ where: { email: String(credentials.email) } })
        if (!user) return null
        const valid = await compare(String(credentials.password), user.passwordHash)
        if (!valid) return null
        return { id: user.id, name: user.name, email: user.email, role: user.role, scopeType: user.scopeType }
      },
    }),
  ],
  callbacks: {
    jwt({ token, user }) {
      if (user) { token.id = user.id; token.role = user.role; token.scopeType = user.scopeType }
      return token
    },
    session({ session, token }) {
      session.user.id = token.id as string
      session.user.role = token.role as Role
      session.user.scopeType = token.scopeType as ScopeType
      return session
    },
  },
})
```

export async function loginAction(email: string, password: string) {
  try {
    await signIn("credentials", { email, password, redirectTo: "/overview" })

import { auth } from "@/lib/auth"
import { redirect } from "next/navigation"
import type { Session } from "next-auth"
import type { Prisma } from "@prisma/client"

export async function requireSession(): Promise<Session> {
  const session = await auth()
  if (!session) redirect("/login")
  return session
}

export async function requireAdmin(): Promise<Session> {
  const session = await requireSession()
  if (session.user.role !== "ADMIN") redirect("/overview")
  return session
}

export function canWrite(session: Session): boolean {
  return session.user.role === "ADMIN" || session.user.role === "DATA_ENTRY"
}

export function donorFilter(session: Session): Prisma.DonorWhereInput {
  if (session.user.scopeType === "ALL") return {}
  return { userScopes: { some: { userId: session.user.id } } }
}

export function recipientFilter(session: Session): Prisma.RecipientWhereInput {
  if (session.user.scopeType === "ALL") return {}
  return { userScopes: { some: { userId: session.user.id } } }
}

import { db } from "@/lib/db"

export async function getPoolBalance(): Promise<number> {
  const [pooled, distributed] = await Promise.all([
    db.donation.aggregate({ _sum: { amount: true }, where: { type: "POOLED" } }),
    db.distribution.aggregate({ _sum: { amount: true }, where: { donationId: null } }),
  ])
  return Number(pooled._sum.amount ?? 0) - Number(distributed._sum.amount ?? 0)
}

export async function getDesignatedUnspent(): Promise<number> {
  const donations = await db.donation.findMany({
    where: { type: "DESIGNATED" },
    include: { distributions: { select: { amount: true } } },
  })
  return donations.reduce((sum, d) => {
    const spent = d.distributions.reduce((s, dist) => s + Number(dist.amount), 0)
    return sum + Math.max(0, Number(d.amount) - spent)
  }, 0)
}

export async function getDonationRemainingBalance(donationId: string): Promise<number> {
  const donation = await db.donation.findUniqueOrThrow({
    where: { id: donationId },
    include: { distributions: { select: { amount: true } } },
  })
  const spent = donation.distributions.reduce((s, d) => s + Number(d.amount), 0)
  return Math.max(0, Number(donation.amount) - spent)
}

"use server"
import { db } from "@/lib/db"
import { requireSession, canWrite, recipientFilter } from "@/lib/permissions"
import { getDonationRemainingBalance } from "@/lib/balance"
import { revalidatePath } from "next/cache"
import { z } from "zod"

const DistributionSchema = z.object({
  recipientId: z.string().uuid(),
  amount: z.coerce.number().positive(),
  date: z.string().date(),
  purpose: z.string().min(1),
  donationId: z.string().uuid().optional(),
  notes: z.string().optional(),
})

export async function createDistribution(data: z.infer<typeof DistributionSchema>) {
  const session = await requireSession()
  if (!canWrite(session)) throw new Error("Unauthorized")
  const parsed = DistributionSchema.parse(data)

  if (parsed.donationId) {
    const remaining = await getDonationRemainingBalance(parsed.donationId)
    if (parsed.amount > remaining) {
      throw new Error(`Amount PKR ${parsed.amount} exceeds remaining balance of PKR ${remaining} on this donation`)
    }
  }

  await db.distribution.create({
    data: {
      recipientId: parsed.recipientId,
      amount: parsed.amount,
      date: new Date(parsed.date),
      purpose: parsed.purpose,
      donationId: parsed.donationId ?? null,
      notes: parsed.notes,
      recordedById: session.user.id,
    },
  })
  revalidatePath("/distributions")
  revalidatePath("/overview")
}

export async function getDistributions() {
  const session = await requireSession()
  const scopeWhere = recipientFilter(session)
  return db.distribution.findMany({
    where: { recipient: Object.keys(scopeWhere).length ? scopeWhere : undefined },
    include: {
      recipient: true,
      donation: { select: { paymentContext: true, donor: { select: { name: true } } } },
      recordedBy: { select: { name: true } },
    },
    orderBy: { date: "desc" },
  })
}

export async function getDesignatedDonationsForRecipient(recipientId: string) {
  await requireSession()
  const donations = await db.donation.findMany({
    where: { type: "DESIGNATED", designatedRecipientId: recipientId },
    include: { distributions: { select: { amount: true } } },
  })
  return donations.map((d) => {
    const spent = d.distributions.reduce((s, dist) => s + Number(dist.amount), 0)
    const remaining = Math.max(0, Number(d.amount) - spent)
    return { id: d.id, amount: Number(d.amount), remaining, date: d.date }
  }).filter((d) => d.remaining > 0)
}

import { db } from "@/lib/db"
import { requireAdmin } from "@/lib/permissions"
import { revalidatePath } from "next/cache"
import { hash } from "bcryptjs"
import { z } from "zod"

const UserSchema = z.object({
  name: z.string().min(1),
  email: z.string().email(),
  password: z.string().min(8).optional(),
  role: z.enum(["ADMIN","DATA_ENTRY","VIEWER"]),
  scopeType: z.enum(["ALL","RESTRICTED"]),
})

export async function createUser(data: z.infer<typeof UserSchema>) {
  await requireAdmin()
  const parsed = UserSchema.parse(data)
  if (!parsed.password) throw new Error("Password required for new users")
  const passwordHash = await hash(parsed.password, 12)
  await db.user.create({
    data: { name: parsed.name, email: parsed.email, passwordHash, role: parsed.role, scopeType: parsed.scopeType },
  })
  revalidatePath("/users")
}

export async function updateUser(id: string, data: Omit<z.infer<typeof UserSchema>, "password"> & { password?: string }) {
  await requireAdmin()
  const updateData: any = { name: data.name, email: data.email, role: data.role, scopeType: data.scopeType }
  if (data.password) updateData.passwordHash = await hash(data.password, 12)
  await db.user.update({ where: { id }, data: updateData })
  revalidatePath("/users")
}

export async function setUserScope(userId: string, donorIds: string[], recipientIds: string[]) {
  await requireAdmin()
  await db.userScope.deleteMany({ where: { userId } })
  const rows = [
    ...donorIds.map((donorId) => ({ userId, donorId })),
    ...recipientIds.map((recipientId) => ({ userId, recipientId })),
  ]
  if (rows.length) await db.userScope.createMany({ data: rows })
  revalidatePath(`/users/${userId}`)
}

export async function getUsers() {
  await requireAdmin()
  return db.user.findMany({ orderBy: { createdAt: "asc" }, include: { _count: { select: { userScopes: true } } } })
}

export async function getUserWithScope(id: string) {
  await requireAdmin()
  return db.user.findUnique({
    where: { id },
    include: {
      userScopes: { include: { donor: { select: { id: true, name: true } }, recipient: { select: { id: true, name: true } } } },
    },
  })
}

const ChangePasswordSchema = z.object({

})

export async function changePassword(data: z.infer<typeof ChangePasswordSchema>) {
  const session = await requireSession()
  const parsed = ChangePasswordSchema.parse(data)
  const user = await db.user.findUniqueOrThrow({ where: { id: session.user.id } })
  const valid = await compare(parsed.currentPassword, user.passwordHash)
  if (!valid) return { error: "Current password is incorrect" }
  await db.user.update({ where: { id: session.user.id }, data: { passwordHash: await hash(parsed.newPassword, 12) } })
  return { success: true }
}

import { NextResponse } from "next/server"
import { getDonorStatement } from "../../../actions"
import ExcelJS from "exceljs"

export async function GET(_req: Request, { params }: { params: { id: string } }) {
  const donor = await getDonorStatement(params.id)
  const wb = new ExcelJS.Workbook()
  const ws = wb.addWorksheet("Donor Statement")

  ws.columns = [
    { header: "Donation Date", key: "donDate", width: 14 },
    { header: "Amount (PKR)", key: "donAmount", width: 14 },
    { header: "Context", key: "context", width: 12 },
    { header: "Type", key: "type", width: 12 },
    { header: "Recipient", key: "recipient", width: 20 },
    { header: "Distributed (PKR)", key: "distAmount", width: 16 },
    { header: "Distribution Date", key: "distDate", width: 14 },
    { header: "Purpose", key: "purpose", width: 30 },
  ]

  for (const donation of donor.donations) {
    if (donation.distributions.length === 0) {
      ws.addRow({
        donDate: donation.date.toLocaleDateString(),
        donAmount: Number(donation.amount),
        context: donation.paymentContext,
        type: donation.type,
        recipient: "", distAmount: "", distDate: "", purpose: "",
      })
    }
    for (const d of donation.distributions) {
      ws.addRow({
        donDate: donation.date.toLocaleDateString(),
        donAmount: Number(donation.amount),
        context: donation.paymentContext,
        type: donation.type,
        recipient: d.recipient.name,
        distAmount: Number(d.amount),
        distDate: d.date.toLocaleDateString(),
        purpose: d.purpose,
      })
    }
  }

  ws.getRow(1).font = { bold: true }
  const buffer = await wb.xlsx.writeBuffer()
  return new NextResponse(buffer as ArrayBuffer, {
    headers: {
      "Content-Type": "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
      "Content-Disposition": `attachment; filename="donor-statement-${params.id}.xlsx"`,
    },
  })
}
